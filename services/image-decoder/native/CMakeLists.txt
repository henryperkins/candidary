cmake_minimum_required(VERSION 3.20)
project(candidary_raw_decoder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
pkg_check_modules(JXL REQUIRED IMPORTED_TARGET libjxl libjxl_threads)
pkg_check_modules(JPEG REQUIRED IMPORTED_TARGET libjpeg)
pkg_check_modules(ZLIB REQUIRED IMPORTED_TARGET zlib)
pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2)
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS "${DNG_SOURCE}/*.cpp")
# Metadata rewriting and the CLI are unused; metadata rewriting requires XMP.
list(FILTER SDK_SOURCES EXCLUDE REGEX "/dng_(validate|xmp_sdk|update_meta)\\.cpp$")
list(REMOVE_ITEM SDK_SOURCES "${DNG_SOURCE}/dng_jxl.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/patch_dng.py"
  "${DNG_SOURCE}/dng_jxl.cpp" "${CMAKE_CURRENT_BINARY_DIR}/dng_jxl.cpp" COMMAND_ERROR_IS_FATAL ANY)
list(APPEND SDK_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/dng_jxl.cpp")
add_library(dngsdk SHARED ${SDK_SOURCES})
target_include_directories(dngsdk PUBLIC "${DNG_SOURCE}")
target_compile_definitions(dngsdk PUBLIC qLinux=1 qDNGUseXMP=0 qDNGUseLibJPEG=1 qDNGSupportVC5=0 qDNGPrintMessages=0 qDNGVerboseExceptions=0 qDNGThreadSafe=1)
target_link_libraries(dngsdk PUBLIC PkgConfig::JXL PkgConfig::JPEG PkgConfig::ZLIB Threads::Threads)
# Use upstream's normal library list; *_ph.cpp files are mutually exclusive stubs.
file(READ "${LIBRAW_SOURCE}/Makefile.am" RAW_MAKEFILE)
string(REGEX MATCHALL "src/[a-zA-Z0-9_./-]+\\.cpp" RAW_SOURCES "${RAW_MAKEFILE}")
list(TRANSFORM RAW_SOURCES PREPEND "${LIBRAW_SOURCE}/")
add_library(raw_r SHARED ${RAW_SOURCES})
target_include_directories(raw_r PUBLIC "${LIBRAW_SOURCE}")
target_compile_definitions(raw_r PRIVATE USE_DNGSDK USE_JPEG USE_JPEG8 USE_ZLIB USE_LCMS2)
target_link_libraries(raw_r PRIVATE dngsdk PkgConfig::JPEG PkgConfig::ZLIB PkgConfig::LCMS Threads::Threads)
add_executable(decode_raw decode_raw.cpp)
target_compile_definitions(decode_raw PRIVATE USE_DNGSDK LIBRAW_DNGSDK_CONFLICT=1)
target_link_libraries(decode_raw PRIVATE raw_r dngsdk)
set_target_properties(decode_raw raw_r dngsdk PROPERTIES INSTALL_RPATH "/opt/decoder/lib")
install(TARGETS decode_raw raw_r dngsdk RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
